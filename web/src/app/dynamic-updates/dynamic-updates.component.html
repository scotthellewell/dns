MikroTik RouterOS Configuration

Example configuration for MikroTik DHCP server to send DNS updates

/ip/dns
set allow-remote-requests=no

/ip/dhcp-server
set [find name="dhcp1"] script=""
set [find name="dhcp1"] lease-script="
  :if ($leaseBound = 1) do={{ '{' }}
    :log info \"DHCP: Lease bound $leaseActMAC -> $leaseActIP\"
    /ip/dns/static/add name=\"$\"leaseActMAC\"\" address=$leaseActIP disabled=no
  {{ '}' }} else={{ '{' }}
    :log info \"DHCP: Lease released $leaseActMAC\"
    /ip/dns/static/remove [find name=\"$\"leaseActMAC\"\"]
  {{ '}' }}
"

info For RFC 2136 Updates, enable Dynamic DNS in MikroTik under IP → DHCP Server → DHCP Tab with your DNS server address.
